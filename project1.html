print('Welcome to Fresh from the Kitchen! We are ready to serve you.\n')

#iprint ang menu
while True:
    print('CHOICES')
    print('\t1. DRINKS')
    print('\t2. RICE MEAL')

    try:
        choice = int(input('\nEnter your choice: '))

        if choice in [1, 2]:
            break
        else:
            print('Invalid main menu choice. Please try again.\n')

    except ValueError:
        print('Please enter a number only.\n')

#pag drinks ang pinili
if choice == 1:

    while True:
        print('Choose your drinks:')
        print('\tA. SOFT DRINKS')
        print('\tB. JUICE')
        print('\tC. ICED TEA')

        drinks = input('\nEnter your choice: ').lower()

        if drinks in ['a', 'b', 'c']:
            break
        else:
            print('Invalid drink choice. Please try again.\n')

    while True:
        print('Size you prefer:')
        print('\tA. REGULAR')
        print('\tB. MEDIUM')

        size = input('\nEnter your choice: ').lower()

        if size in ['a', 'b']:
            break
        else:
            print('Invalid size choice. Please try again.\n')

    if drinks == 'a':
        if size == 'a':
            print('Your order is a Regular Soft Drink')
        else:
            print('Your order is a Medium Soft Drink')

    elif drinks == 'b':
        if size == 'a':
            print('Your order is a Regular Juice')
        else:
            print('Your order is a Medium Juice')

    elif drinks == 'c':
        if size == 'a':
            print('Your order is a Regular Iced Tea')
        else:
            print('Your order is a Medium Iced Tea')
