Hi! This is the code.

To make it work, install XAMPP or WAMP. Copy+paste and save them in separate .php files under the same folder. Place file in the root folder, e.g. 'C:\xampp\htdocs\'. Open server control panel and click Start next to Apache. Then paste its location from htdocs folder like this: 'http://localhost/filename.php' onto the browser address bar. You can also paste the code directly to online sandboxes like OnlinePHP.io or Jumpshare or use VSCode or the built-in PHP server if PHP is installed.

Book.php

<?php

class Book{
  public $bookTitle;
  public $author;
  public $borrowerName = null;
  public $daysBorrowed = 0;

  function set_details($bookTitle, $author, $borrowerName, $daysBorrowed) {
    $this->bookTitle = $bookTitle;
    $this->author = $author;
    $this->borrowerName = $borrowerName;
    $this->daysBorrowed = $daysBorrowed;
  }
  function get_details() {
    echo "Book Title: " . $this->bookTitle . "
";
    echo "Author: " . $this->author . "
";
    echo "Borrower's Name: " . $this->borrowerName ."
";
    echo "Days Borrowed: " . $this->daysBorrowed . "
";
  }
}

//calculateLateFee to sum up the borrower's fees
function calculateLateFee($daysBorrowed){
    $lateFee = 0;

if ($daysBorrowed == 0) {
    echo "Book available for borrow, no late fees
";
}
else {
    $i = 8;

    while ($i <= $daysBorrowed) {
        if ($i <= 14) {
            $lateFee += 10;
        } else {
            $lateFee += 20;
        }
        $i++;
    }

    echo "Late Fee: ₱" . $lateFee . "
";
}
}

//The getStatus function
function getStatus($borrowerName) {
    if ($borrowerName == null) {
        echo "Status: Available

";
    } else {
        echo "Status: Borrowed

";
    }
}
?>

index.php

<?php
include 'Book.php';
include 'libraryData.php';

//Display books using a loop

for($i = 0; $i < count($books); $i++) {
    $books[$i]->get_details();
    calculateLateFee($books[$i]->daysBorrowed);
    getStatus($books[$i]->borrowerName);
}
